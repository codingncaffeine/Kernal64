# SOURCE  DEFINES  MODEL (as k64run --model)  CIA  EXPECTED (the reference emulator's value)
# Serial-port interrupt: timer B's low byte at the handler, two alignments, both chips.
sdrirq.s  SHIFT=0,MASK=$88  pal  old  169
sdrirq.s  SHIFT=1,MASK=$88  pal  old  170
sdrirq.s  SHIFT=0,MASK=$88  pal  new  171
sdrirq.s  SHIFT=1,MASK=$88  pal  new  170
# Light pen interrupt flag: 3 = set by any trigger, 2 = only for a line held low over the frame start.
lpirq.s  -  pal  old  3
lpirq.s  -  pal-new  old  3
lpirq.s  -  ntsc  old  3
lpirq.s  -  ntsc-new  old  3
lpirq.s  -  pal-old  old  2
lpirq.s  -  ntsc-old  old  2
# Paddle registers and the switch to the control ports (our own rule: the reference has no
# paddles to set from its command line). Paddles in both ports: every step reads its value.
# Without them, or with one pair, the first step that expects a paddle fails.
pots.s  X1=63,Y1=10,X2=191,Y2=200,XB=47,YB=10  pal  old  0  --paddles 1 63 10 --paddles 2 191 200
pots.s  X1=63,Y1=10,X2=191,Y2=200,XB=47,YB=10  ntsc  new  0  --paddles 1 63 10 --paddles 2 191 200
pots.s  X1=63,Y1=10,X2=191,Y2=200,XB=47,YB=10  pal  old  2
pots.s  X1=63,Y1=10,X2=191,Y2=200,XB=47,YB=10  pal  old  3  --paddles 1 63 10
pots.s  X1=63,Y1=10,X2=191,Y2=200,XB=47,YB=10  pal  old  2  --paddles 2 191 200
pots.s  X1=63,Y1=10,X2=191,Y2=200,XB=48,YB=10  pal  old  4  --paddles 1 63 10 --paddles 2 191 200
# A program started by the machine's autostart finds BASIC's pointers as LOAD leaves them.
autostart.s  -  pal  old  0
autostart.s  -  ntsc  new  0
